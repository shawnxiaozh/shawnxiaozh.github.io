## [543. Diameter of Binary Tree](https://leetcode.cn/problems/diameter-of-binary-tree/)
```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    int ans = Integer.MIN_VALUE;
    public int diameterOfBinaryTree(TreeNode root) {
      dfs(root);
      return ans;
    }

    private int dfs(TreeNode root) {
      if(root == null) {
        return -1;
      }
      int lLen = dfs(root.left);
      int rLen = dfs(root.right);

      ans = Math.max(ans, lLen + rLen + 2);
      return Math.max(lLen, rLen) + 1;
    }
}

```

```java
      dfs(root);
      return ans;
    }

    // dfs(root) := root为根结点的 最大
    private int dfs(TreeNode root) {
      if(root == null) {
        return -1;
      }
      int lLen = dfs(root.left) + 1;
      int rLen = dfs(root.right) + 1;

      ans = Math.max(ans, lLen + rLen);
      return Math.max(lLen, rLen);
    }
}
```

## [124. Binary Tree Maximum Path Sum](https://leetcode.cn/problems/binary-tree-maximum-path-sum/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    int ans = Integer.MIN_VALUE;
    public int maxPathSum(TreeNode root) {
      dfs(root);
      return ans;
    }
    // dfs(root) = df
    private int dfs(TreeNode root) {
      if(root == null) {
        return 0;
      }
      int lSum = Math.max(dfs(root.left), 0);
      int rSum = Math.max(dfs(root.right), 0);

      ans = Math.max(ans, lSum + rSum + root.val);
      return Math.max(lSum, rSum) + root.val;
    }
}
```

## [1245. Tree Diameter](https://leetcode.cn/problems/tree-diameter/)
```java
class Solution {
    int ans = 0;
    public int treeDiameter(int[][] edges) {
      // build graph
      Map<Integer, List<Integer>> graph = new HashMap<>();
      for(int[] edge : edges) {
        int u = edge[0];
        int v = edge[1];
        graph.computeIfAbsent(u, k -> new ArrayList<>()).add(v);
        graph.computeIfAbsent(v, k -> new ArrayList<>()).add(u);
      }
      if(graph.size() == 0) {
        return 0;
      }
      boolean[] visited = new boolean[graph.size()];
      // 随机选一个结点都可以作为根结点开始
      visited[1] = true;
      dfs(1, graph, visited);
      return ans;
    }


    // dfs(u) := 以u为根，从u到最长的叶子结点的深度
    private int dfs(int u, Map<Integer, List<Integer>> graph, boolean[] visited) {
      int max = 0;
      int second = 0;
      
      // v 为子结点
      for(int v : graph.get(u)) {
        if(!visited[v]) {
          // 这里邻接边除去父节点，最多2条，一条左边，一条右边
          // 所以这里有max和second
          visited[v] = true;
          // 以v为根，从v到最长的叶子结点的深度
          int ret = dfs(v, graph, visited);
          // System.out.println("cur: " + u + "   next: " + v + "    ret:" + ret);
          if(ret > max) {
            second = max;
            // 最大深度
            max = ret;
          } else if(ret > second) {
            // 最小深度
            second = ret;
          }
          // v结点的直径为左子树深度+ 右子树深度
          ans = Math.max(ans, max + second);
        }
      }
      // 走到了叶子结点u，深度为1
      return max + 1;
    }
}
```

## [834. Sum of Distances in Tree](https://leetcode.cn/problems/sum-of-distances-in-tree/)
[力扣](https://leetcode.cn/problems/sum-of-distances-in-tree/)

```java
class Solution {
  private List<Integer>[] graph;
  private int[] ans;
  private int[] size;
  public int[] sumOfDistancesInTree(int n, int[][] edges) {

    // ans[v] = ans[u] + n - size(v) - size(v) = ans[u] + n - 2 * size(v)
    // size(v) := number of nodes in the root of v
    // n - size(v) := number of nodes out the root of v
    graph = new ArrayList[n];
    Arrays.setAll(graph, e -> new ArrayList<>());
    for(int[] edge : edges) {
      int u = edge[0], v = edge[1];
      graph[u].add(v);
      graph[v].add(u);
    }
    ans = new int[n];
    size = new int[n];
    dfs(0, -1, 0);
    reroot(0, -1);
    return ans;
  }

  private void dfs(int u, int father, int depth) {
    ans[0] += depth;
    size[u] = 1;
    for(int v : graph[u]) {
      if(v != father) {
        dfs(v, u, depth + 1);
        size[u] += size[v];
      }
    }
  }

  private void reroot(int u, int father) {
    for(int v : graph[u]) {
      if(v != father) {
        ans[v] = ans[u] + graph.length - 2 * size[v];
        reroot(v, u);
      }
    }
  }
}
```
