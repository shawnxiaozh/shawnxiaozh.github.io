## [2741. Special Permutations](https://leetcode.cn/problems/special-permutations/)
dfs(i, j) 定义为可以选的下标集合为i，上一个数选的下标为j时，可以构造出多少个特别的排列。

枚举当前要选的数的下标k，如果nums[k]与nums[j]满足题目整除的要求，则有

$ dfs(i,j)=\sum_{k\in i}(dfs(i \setminus {k}, k)) $

递归边界：$ dfs(0,j)=1 $，表示找到了一个特别排列。

递归入口：nums中的每一个数都当第一个数$ dfs(U \setminus j,j) $时 ,其中全集为 $ U={1, 2, ..., n-1} $，累计的特殊排列数。

```python
class Solution:
    def specialPerm(self, nums: List[int]) -> int:
      MOD = 10 ** 9 + 1;
      n = len(nums)
      @cache
      def dfs(i: int, j: int) -> int:
        if i == 0:
          return 1
        res = 0
        for k, x in enumerate(nums):
          # if k in collection i: (i >> k & i) == 1 is true
          if i >> k & 1 and (nums[j] % x == 0 or x % nums[j] == 0):
            # delete k: i ^ (1 << k)
            res += dfs(i ^ (1 << k), k)
        return res
      # (1 << n) - 1: 全集
      u = (1 << n) - 1
      
      return sum(dfs(u ^ (1 << i), i) for i in range(n)) % MOD
```

```java
class Solution {

  int MOD = (int) 1e9 + 7;
  public int specialPerm(int[] nums) {
    int[][] dp = new int[1 << nums.length][nums.length + 1];
    for(int[] dpi : dp) {
      Arrays.fill(dpi, -1);
    }
    int res = 0;
 		// bits: nums中所有的数映射的二进制集合。 -的优先级大于<<
    int bits = (1 << nums.length) - 1;
    for(int i = 0; i < nums.length; i++) {
      // bits ^ (1 << i) bits集合去除i
      res = (res + dfs(nums, bits ^ (1 << i), i, dp)) % MOD;
    }
    return res;
  }

  private int dfs(int[] nums, int bits, int j, int[][] dp) {
    if(bits == 0) {
      return 1;
    }
    // 以 prevPos 和 二进制占位为key，只要二进制占位相同，数组长度也必然是相同的。
    if (dp[bits][j] != -1) {
      return dp[bits][j];
    }

    int res = 0;
    for(int i = 0; i < nums.length; i++) {
      // (bits >> i & 1) == 1： 判断i是否在bits集合中
      if(((bits >> i & 1) == 1) && (nums[j] % nums[i] == 0 || nums[i] % nums[j] == 0)) {
        	// // bits ^ (1 << i) bits集合去除i
          res = (res + dfs(nums, bits ^ (1 << i), i, dp)) % MOD;
      }
    }
    return dp[bits][j] = res;
  }
}
```

时间复杂度：O(状态个数) * O(单个状态的计算时间) = O(n * 2^n) * O(n) = O(n^2*2^n)

$ dp[i][j]=\sum_{k\in i}(dp[i \setminus {k}][k]) $

$ dp[u | 1 << j][k]= \sum_{k \in u}{dp[u][k]}  $
