## [64. Minimum Path Sum](https://leetcode.cn/problems/minimum-path-sum/)
```java
class Solution {
    public int minPathSum(int[][] grid) {
      
      // dp[i][j] = min(dp[i-1][j], dp[i][j-1]) + grid[i][j]
      int[][] dp = new int[grid.length][grid[0].length];
      dp[0][0] = grid[0][0];
      for(int j = 1; j < grid[0].length; j++) {
        dp[0][j] = dp[0][j - 1] + grid[0][j];
      }
      for(int i = 1; i < grid.length; i++) {
        dp[i][0] = dp[i - 1][0] + grid[i][0];
      }

      for(int i = 1; i < grid.length; i++) {
        for(int j = 1; j < grid[0].length; j++) {
          dp[i][j] = Math.min(dp[i][j - 1], dp[i - 1][j]) + grid[i][j];
        }
      }
      return dp[grid.length - 1][grid[0].length - 1];

    }
}
```

## [221. Maximal Square](https://leetcode.cn/problems/maximal-square/)

```java
class Solution {
  public int maximalSquare(char[][] matrix) {
    // 我们用 dp(i,j) 表示以 (i,j) 为右下角，且只包含 1 的正方形的边长最大值。如果我们能计算出所有 
    // dp(i,j) 的值，那么其中的最大值即为矩阵中只包含 1 的正方形的边长最大值，其平方即为最大正方形的面积
    // dp[i][j] = min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1]) + 1
    int[][] dp = new int[matrix.length + 1][matrix[0].length + 1];
    int maxEdge = 0;
    for(int i = 0; i < matrix.length; i++) {
      for(int j = 0; j < matrix[0].length; j++) {
        if(matrix[i][j] == '1') {
          dp[i + 1][j + 1] = Math.min(dp[i][j + 1], Math.min(dp[i + 1][j], dp[i][j])) + 1;
          maxEdge = Math.max(maxEdge, dp[i + 1][j + 1]);
        }
      }
    }


    return maxEdge * maxEdge;
  }
}
```

## [446. 等差数列划分 II - 子序列](https://leetcode.cn/problems/arithmetic-slices-ii-subsequence/)
```java
class Solution {
    public int numberOfArithmeticSlices(int[] nums) {
        // dp[i][d] := 以nums[i]结尾，公差为d的子序列个数
        // dp[i][d] = sum(dp[j][d] + 1)
        int n = nums.length;

        List<Map<Long, Integer>> list = new ArrayList<>();
        // [{{d1, cnt1}, {d2, cnt2}}, {d1, cnt3}]
        for(int i = 0; i < n; i++) {
            Map<Long, Integer> cur = new HashMap<>();
            for(int j = 0; j < i; j++) {
                // 公差
                Long d = nums[i] * 1L - nums[j];
                // sum(dp[j][d] + 1)
                int sum = list.get(j).getOrDefault(d, 0) + 1;
                cur.put(d, cur.getOrDefault(d, 0) + sum);
            }
            list.add(cur);
        }

        // 统计的都是长度大于1的等差子序列数量，长度为0的不会在map中出现
        int ans = 0;
        for(int i = 0; i < n; i++) {
            ans += list.get(i).values().stream().mapToInt(p -> p.intValue()).sum();
        }
        // 长度至少为3的 = ans - 减去长度为2的等差子序列
        // 任意的两个数天然都是长度为2的等差子序列，求出两个数能组成的对数就是 「减去长度为2的等差子序列」数目
        // (n - 1) + (n - 2) + .. + 1  = (1 + n - 1) * (n - 1) / 2
        return ans - n * (n - 1) /2;
    }
}
```

## [面试题 17.24. Max Submatrix LCCI](https://leetcode.cn/problems/max-submatrix-lcci/)
```java
class Solution {
  public int[] getMaxMatrix(int[][] matrix) {


    int max = Integer.MIN_VALUE;
    int[] res = new int[4];
    for(int startRow = 0; startRow < matrix.length; startRow++) {
      int[] nums = new int[matrix[0].length];
      // Arrays.copyOfRange(matrix[startRow], 0, matrix.length);
      for(int endRow = startRow; endRow < matrix.length; endRow++) {
        for(int j = 0; j < matrix[0].length; j++) {
          nums[j] = nums[j] + matrix[endRow][j];
        }
        int[] submax = getMaxSubArray(nums);
        if(max < submax[0]) {
          max = submax[0];
          res[0] = startRow;
          res[1] = submax[1];
          res[2] = endRow;
          res[3] = submax[2];
          // [1, -3]
          // [-4, 9]
          // System.out.println(max + " " + submax[1] + " " + submax[2]);
        }
      }
    }
    return res;
  }

  // 一维的最大子矩阵和
  private int[] getMaxSubArray(int[] nums) {
    int[] preSum = new int[nums.length + 1];
    // dp[i] = min(dp[i - 1] + nums[i], nums[i])
    int max = Integer.MIN_VALUE;
    int[] res = new int[3];
    for(int i = 0; i < nums.length; i++) {
      preSum[i + 1] = preSum[i] + nums[i];
    }
    TreeMap<Integer, Integer> treeMap = new TreeMap<>();
    treeMap.put(0, -1);
    for(int i = 0; i < nums.length; i++) {
      int rangeSum = preSum[i + 1] - treeMap.firstKey();
      if (rangeSum > max) {
        max = rangeSum;
        res[0] = max;
        res[1] = treeMap.get(treeMap.firstKey()) + 1;
        res[2] = i;
      }
      treeMap.putIfAbsent(preSum[i + 1], i);
    }
    return res;
  }
      // 一维的最大子矩阵和
    private static int[] getMaxSubArray(int[] nums) {
      int maxsum = Integer.MIN_VALUE;
      int[] dp = new int[nums.length + 1];
      int[] ans = new int[3];
      int begin = 0;
      for(int i = 0; i < nums.length ; i++ ){
        // dp[i + 1] = Math.min(dp[i] + nums[i], nums[i]) = Math.max(dp[i], 0) + nums[i]
          if(dp[i] > 0) {  
            dp[i + 1] = dp[i] + nums[i];
          } else {              
            dp[i + 1] = nums[i];
            begin = i;     
          }
          if(dp[i + 1] > maxsum){
              maxsum = dp[i + 1];
              ans[0] = maxsum;
              ans[1] = begin;//记录下起始和终止位置
              ans[2] = i;
          }  
      }
      return ans;
    }
  
}
```
